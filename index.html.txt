<script>
    // Standard Spiel-Status
    let defaultPet = {
        hunger: 100,
        joy: 100,
        energy: 100,
        health: 100,
        coins: 0,
        level: 1,
        xp: 0,
        isSleeping: false,
        isDead: false
    };

    // Aktuelles Pet (wird beim Laden überschrieben, falls ein Savegame existiert)
    let pet = { ...defaultPet };

    const MAX_STAT = 100;
    const XP_PER_LEVEL = 100;

    // --- SPEICHER-FUNKTIONEN ---

    function saveGame() {
        if (pet.isDead) return; // Tote Pets werden nicht mehr gespeichert
        localStorage.setItem('cyberPetSave', JSON.stringify(pet));
        
        // Kurzes Aufblinken des "Gespeichert"-Icons
        const saveIcon = document.getElementById('save-icon');
        saveIcon.style.opacity = '1';
        setTimeout(() => { saveIcon.style.opacity = '0'; }, 1000);
    }

    function loadGame() {
        const savedData = localStorage.getItem('cyberPetSave');
        if (savedData) {
            pet = JSON.parse(savedData);
            logMessage("Spielstand erfolgreich geladen! Willkommen zurück.");
        } else {
            logMessage("Neues Spiel gestartet. Viel Spaß!");
        }
    }

    function resetGame() {
        if (confirm("Willst du wirklich von vorne anfangen? Dein aktueller Fortschritt geht komplett verloren.")) {
            localStorage.removeItem('cyberPetSave');
            location.reload(); // Lädt die Seite neu
        }
    }

    // --- SPIEL-LOGIK ---

    function updateUI() {
        if (pet.isDead) return;

        document.getElementById('hunger-fill').style.width = pet.hunger + '%';
        document.getElementById('hunger-val').innerText = Math.round(pet.hunger);
        
        document.getElementById('joy-fill').style.width = pet.joy + '%';
        document.getElementById('joy-val').innerText = Math.round(pet.joy);
        
        document.getElementById('energy-fill').style.width = pet.energy + '%';
        document.getElementById('energy-val').innerText = Math.round(pet.energy);
        
        document.getElementById('health-fill').style.width = pet.health + '%';
        document.getElementById('health-val').innerText = Math.round(pet.health);

        document.getElementById('coins').innerText = pet.coins;
        document.getElementById('level').innerText = pet.level;
        document.getElementById('xp').innerText = pet.xp;

        updateFace();
    }

    function updateFace() {
        const face = document.getElementById('pet-face');
        if (pet.isDead) {
            face.innerText = "💀";
            return;
        }
        if (pet.isSleeping) {
            face.innerText = "😴";
            return;
        }
        
        if (pet.health < 30) face.innerText = "🤢";
        else if (pet.joy < 30) face.innerText = "😢";
        else if (pet.hunger < 30) face.innerText = "😫";
        else if (pet.energy < 30) face.innerText = "🥱";
        else face.innerText = "😊";
    }

    function logMessage(msg) {
        document.getElementById('message-board').innerText = msg;
    }

    function addXP(amount) {
        pet.xp += amount;
        if (pet.xp >= XP_PER_LEVEL) {
            pet.level++;
            pet.xp -= XP_PER_LEVEL;
            logMessage(`🎉 Level Up! Dein Pet ist jetzt Level ${pet.level}!`);
        }
    }

    function feed() {
        if (pet.isSleeping || pet.isDead) return logMessage("Dein Pet kann jetzt nicht essen.");
        pet.hunger = Math.min(pet.hunger + 15, MAX_STAT);
        addXP(5);
        logMessage("Du fütterst dein Pet. Lecker!");
        updateUI();
        saveGame();
    }

    function play() {
        if (pet.isSleeping || pet.isDead) return logMessage("Dein Pet kann jetzt nicht spielen.");
        if (pet.energy < 15) return logMessage("Dein Pet ist zu müde zum Spielen!");
        
        pet.joy = Math.min(pet.joy + 20, MAX_STAT);
        pet.energy -= 15;
        pet.hunger -= 10;
        addXP(10);
        logMessage("Ihr spielt zusammen. Das macht Spaß!");
        updateUI();
        saveGame();
    }

    function sleep() {
        if (pet.isDead) return;
        pet.isSleeping = true;
        logMessage("Dein Pet schläft jetzt. Zzz...");
        updateUI();
        saveGame();
        
        setTimeout(() => {
            if (pet.isDead) return;
            pet.isSleeping = false;
            pet.energy = MAX_STAT;
            logMessage("Dein Pet ist aufgewacht und voller Energie!");
            updateUI();
            saveGame();
        }, 5000); // Schläft für 5 Sekunden
    }

    function work() {
        if (pet.isSleeping || pet.isDead) return logMessage("Dein Pet kann jetzt nicht arbeiten.");
        if (pet.energy < 25) return logMessage("Dein Pet ist zu erschöpft für die Arbeit.");
        if (pet.joy < 20) return logMessage("Dein Pet ist zu traurig zum Arbeiten.");

        pet.energy -= 25;
        pet.joy -= 10;
        pet.hunger -= 15;
        let earned = Math.floor(Math.random() * 5) + 3;
        pet.coins += earned;
        addXP(15);
        
        logMessage(`Dein Pet hat gearbeitet und ${earned} 💰 verdient!`);
        updateUI();
        saveGame();
    }

    function heal() {
        if (pet.isDead) return;
        if (pet.health >= MAX_STAT) return logMessage("Dein Pet ist bereits kerngesund.");
        pet.health = Math.min(pet.health + 30, MAX_STAT);
        pet.joy -= 5;
        logMessage("Du gibst deinem Pet Medizin. Es fühlt sich besser.");
        updateUI();
        saveGame();
    }

    function buyPremiumFood() {
        if (pet.isDead) return;
        if (pet.coins >= 10) {
            pet.coins -= 10;
            pet.hunger = Math.min(pet.hunger + 40, MAX_STAT);
            pet.joy = Math.min(pet.joy + 10, MAX_STAT);
            addXP(20);
            logMessage("Premium-Pizza gekauft! Dein Pet liebt es!");
            updateUI();
            saveGame();
        } else {
            logMessage("Nicht genug Münzen! Geh arbeiten.");
        }
    }

    // --- GAME LOOP ---
    
    setInterval(() => {
        if (pet.isDead) return;

        // Werte sinken mit der Zeit
        if (!pet.isSleeping) {
            pet.hunger -= 2;
            pet.joy -= 1.5;
            pet.energy -= 1;
        }

        // Wenn Hunger oder Freude zu niedrig, leidet die Gesundheit
        if (pet.hunger < 20 || pet.joy < 20) {
            pet.health -= 3;
        }

        // Grenzen einhalten
        pet.hunger = Math.max(pet.hunger, 0);
        pet.joy = Math.max(pet.joy, 0);
        pet.energy = Math.max(pet.energy, 0);
        pet.health = Math.max(pet.health, 0);

        // Check auf Tod
        if (pet.health <= 0) {
            pet.isDead = true;
            localStorage.removeItem('cyberPetSave'); // Lösche den Spielstand
            logMessage("Oh nein! Dein Pet hat es nicht geschafft. Klicke auf 'Spielstand löschen' für ein neues Pet.");
            updateFace();
            return; // Abbrechen, kein Auto-Save mehr
        }

        updateUI();
        saveGame(); // Automatisches Speichern jeden Tick
    }, 2000); // Alle 2 Sekunden

    // --- INITIALISIERUNG ---
    loadGame(); // Beim Starten der Seite den Spielstand laden
    updateUI(); // Benutzeroberfläche direkt aktualisieren
</script>
